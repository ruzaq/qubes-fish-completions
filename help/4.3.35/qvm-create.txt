usage: qvm-create [--verbose] [--quiet] [--help] [--version] [--class CLS]
                  [--standalone] [--disp] [--property NAME=VALUE]
                  [--pool VOLUME_NAME=POOL_NAME] [-P POOL_NAME]
                  [--template VALUE] [--label VALUE] [--help-classes]
                  [--root-copy-from FILENAME | --root-move-from FILENAME]
                  [VMNAME]

positional arguments:
  VMNAME                name of the domain to create

options:
  --verbose, -v         increase verbosity
  --quiet, -q           decrease verbosity
  --help, -h            show this help message and exit
  --version             show program's version number and exit
  --class, -C CLS       specify the class of the new domain (default: AppVM)
  --standalone          shortcut for --class StandaloneVM
  --disp                alias for --class DispVM --label red
  --property, --prop NAME=VALUE
                        set domain's property, like "internal", "memory" or
                        "vcpus"
  --pool, -p VOLUME_NAME=POOL_NAME
                        specify the pool to use for a volume
  -P POOL_NAME          change all volume pools to specified pool
  --template, -t VALUE  specify the TemplateVM to use
  --label, -l VALUE     specify the label to use for the new domain (e.g. red,
                        yellow, green, ...)
  --help-classes        List available classes and exit
  --root-copy-from, -r FILENAME
                        use provided root.img instead of default/empty one
                        (file will be COPIED)
  --root-move-from, -R FILENAME
                        use provided root.img instead of default/empty one
                        (file will be MOVED)
